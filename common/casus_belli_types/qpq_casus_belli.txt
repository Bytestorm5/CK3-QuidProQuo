# Quid Pro Quo casus belli

#################################################################
# Crown Mandate: a Chartered Conquest lends the liege's de jure right
# to a vassal who has no claim of their own.
# The attacker keeps the titles they were chartered to take in variable list qpq_mandate_titles.
#################################################################
qpq_crown_mandate_cb = {
	icon = de_jure_cb
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		scope:attacker = {
			has_variable_list = qpq_mandate_titles
			is_landed = yes
		}
	}

	allowed_against_character = {
		scope:attacker = {
			any_in_list = {
				variable = qpq_mandate_titles
				holder ?= {
					OR = {
						this = scope:defender
						top_liege = scope:defender
					}
				}
			}
			NOT = { top_liege = scope:defender.top_liege }
		}
	}

	target_titles = all
	target_title_tier = all
	ignore_effect = change_title_holder

	valid_to_start = {
		scope:attacker = {
			is_target_in_variable_list = { name = qpq_mandate_titles target = scope:target }
		}
	}

	should_invalidate = {
		NOT = {
			scope:attacker = {
				any_in_list = {
					variable = qpq_mandate_titles
					holder ?= {
						OR = {
							this = scope:defender
							top_liege = scope:defender
						}
					}
				}
			}
		}
	}

	on_invalidated_desc = qpq_crown_mandate_cb_invalidated_desc
	on_invalidated = {}

	cost = {
		prestige = {
			value = 0
			add = {
				value = 50
				desc = CB_BASE_COST
			}
			multiply = common_cb_prestige_cost_multiplier
		}
	}

	on_declaration = {
		on_declared_war = yes
		qpq_charter_war_declared_effect = yes
	}

	on_victory_desc = {
		desc = qpq_crown_mandate_cb_victory_desc
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			if = {
				limit = { tier = tier_county }
				change_title_holder = {
					holder = scope:attacker
					change = scope:change
				}
			}
			else = {
				every_in_de_jure_hierarchy = {
					limit = {
						tier = tier_county
						holder ?= {
							OR = {
								this = scope:defender
								target_is_liege_or_above = scope:defender
							}
						}
					}
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
				if = {
					limit = { holder ?= scope:defender }
					change_title_holder = {
						holder = scope:attacker
						change = scope:change
					}
				}
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:attacker = {
			add_prestige_experience = minor_prestige_value
		}
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = qpq_crown_mandate_cb_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = {
			add_prestige = minor_prestige_loss
		}
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		desc = qpq_crown_mandate_cb_defeat_desc
	}

	on_defeat = {
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 2
			}
			add_prestige = medium_prestige_loss
		}
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "QPQ_CROWN_MANDATE_WAR_NAME"
	war_name_base = "QPQ_CROWN_MANDATE_WAR_NAME_BASE"
	cb_name = "QPQ_CROWN_MANDATE_CB_NAME"

	interface_priority = 80
	use_de_jure_wargoal_only = yes
	attacker_wargoal_percentage = 0.8
	max_defender_score_from_occupation = 150
	max_attacker_score_from_occupation = 150

	ai_score = {
		value = 100
	}
}

#################################################################
# Avenge Perfidy: the victim of an exposed Tier IV-V pact may go to war against the payer
#################################################################
qpq_avenge_perfidy_cb = {
	icon = humiliation_cb
	group = humiliation

	allow_hostages = yes

	allowed_for_character = {
		scope:attacker = { is_landed = yes }
	}

	allowed_against_character = {
		scope:attacker = {
			has_opinion_modifier = {
				target = scope:defender
				modifier = qpq_avenge_perfidy_opinion
			}
		}
		scope:defender = { is_landed = yes }
	}

	target_titles = none
	ignore_effect = change_title_holder

	should_invalidate = {
		scope:defender = { is_landed = no }
	}

	on_invalidated_desc = qpq_avenge_perfidy_cb_invalidated_desc
	on_invalidated = {}

	cost = {
		prestige = {
			value = 0
		}
	}

	on_declaration = {
		on_declared_war = yes
	}

	on_victory_desc = {
		desc = qpq_avenge_perfidy_cb_victory_desc
	}

	on_victory = {
		scope:attacker = { show_pow_release_message_effect = yes }
		scope:defender = {
			pay_short_term_gold = {
				target = scope:attacker
				gold = 4
				yearly_income = yes
			}
			add_prestige = major_prestige_loss
		}
		scope:attacker = {
			add_prestige = medium_prestige_gain
			add_piety = minor_piety_gain
			if = {
				limit = {
					has_opinion_modifier = {
						target = scope:defender
						modifier = qpq_avenge_perfidy_opinion
					}
				}
				remove_opinion = {
					target = scope:defender
					modifier = qpq_avenge_perfidy_opinion
				}
			}
		}
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = qpq_avenge_perfidy_cb_white_peace_desc
	}

	on_white_peace = {
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		desc = qpq_avenge_perfidy_cb_defeat_desc
	}

	on_defeat = {
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 2
			}
			add_prestige = minor_prestige_loss
		}
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = invalidate
	attacker_allies_inherit = yes
	defender_allies_inherit = yes

	war_name = "QPQ_AVENGE_PERFIDY_WAR_NAME"
	war_name_base = "QPQ_AVENGE_PERFIDY_WAR_NAME_BASE"
	cb_name = "QPQ_AVENGE_PERFIDY_CB_NAME"

	interface_priority = 70
	ticking_war_score_targets_entire_realm = yes

	ai_score = {
		value = 40
	}
}

#################################################################
# Pact War: a war someone else is paying for (Sanctioned Conquest, Proxy War,
# Partition Pact, Licensed Feud, Blessing for a Feud, Holy Warrant).
# The agent keeps a list of characters they are licensed to attack: qpq_war_licence_against
#################################################################
qpq_pact_war_cb = {
	icon = county_conquest_cb
	group = conquest

	combine_into_one = yes
	should_show_war_goal_subview = yes
	mutually_exclusive_titles = { always = yes }

	allowed_for_character = {
		scope:attacker = {
			has_variable_list = qpq_war_licence_against
			is_landed = yes
		}
	}

	allowed_against_character = {
		scope:attacker = {
			is_target_in_variable_list = { name = qpq_war_licence_against target = scope:defender }
		}
		NOR = {
			scope:defender = { target_is_liege_or_above = scope:attacker }
			scope:attacker = { target_is_liege_or_above = scope:defender }
		}
	}

	target_titles = all
	target_title_tier = county
	ignore_effect = change_title_holder

	valid_to_start = {
		scope:target = {
			is_landless_type_title = no
			holder ?= {
				OR = {
					this = scope:defender
					target_is_liege_or_above = scope:defender
				}
			}
			OR = {
				is_neighbor_to_realm = scope:attacker
				scope:attacker = { is_ai = no }
			}
		}
	}

	should_invalidate = {
		NOT = {
			scope:attacker = {
				is_target_in_variable_list = { name = qpq_war_licence_against target = scope:defender }
			}
		}
	}

	on_invalidated_desc = qpq_pact_war_cb_invalidated_desc
	on_invalidated = {}

		prestige = {
			value = 0

				value = 25

		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = yes
		}
		every_in_list = {
			list = target_titles
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
			}
		}
		resolve_title_and_vassal_change = scope:change
		scope:defender = {
			add_prestige = minor_prestige_loss
		}
		scope:attacker = {
			add_prestige_experience = minor_prestige_value
		}
		add_truce_attacker_victory_effect = yes
	}

	on_white_peace_desc = {
		desc = qpq_pact_war_cb_white_peace_desc
	}

	on_white_peace = {
		scope:attacker = {
			add_prestige = minor_prestige_loss
		}
		add_truce_white_peace_effect = yes
	}

	on_defeat_desc = {
		desc = qpq_pact_war_cb_defeat_desc
	}

	on_defeat = {
		scope:attacker = {
			pay_short_term_gold_reparations_effect = {
				GOLD_VALUE = 2
			}
			add_prestige = medium_prestige_loss
		}
		add_truce_attacker_defeat_effect = yes
	}

	transfer_behavior = transfer
	on_primary_attacker_death = inherit
	on_primary_defender_death = inherit
	attacker_allies_inherit = yes
	defender_allies_inherit = yes
