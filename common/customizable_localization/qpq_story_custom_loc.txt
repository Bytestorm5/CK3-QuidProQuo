QpqOwesGoldDue = {
	type = all
	log_loc_errors = no
	text = {
		trigger = {
			exists = var:qpq_main
			exists = var:qpq_gold_due var:qpq_gold_due > 0
		}
		localization_key = qpq_sl_owed_gold_due
	}
	text = {
		trigger = {
			NOT = { exists = var:qpq_main }
			exists = var:qpq_gold_due var:qpq_gold_due > 0
		}
		localization_key = qpq_sl_owe_gold_due
	}
	text = {
		fallback = yes
		localization_key = qpq_empty
	}
}
