# Quid Pro Quo - Backroom Deal panel: can the proposal be sent? (hand-written)
# Root and this: the player. Scopes: actor (the player), recipient.
# Used by the Propose button (as its failure reasons) and again when the proposal is sent,
# after a last refresh, since the world may have moved on while the panel was open.
# Each leg's own checks are generated (qpq_ui_leg_<leg>_ready_trigger in qpq_ui_kind_triggers.txt).

qpq_ui_can_propose_trigger = {
	custom_description = {
		text = QPQ_UI_FAIL_EMPTY
		OR = {
			has_variable = qpq_ui_t_ekind
			has_variable = qpq_ui_i_ekind
		}
	}
	qpq_ui_leg_t_ready_trigger = yes
	qpq_ui_leg_i_ready_trigger = yes
	# A deal of two deeds makes two pacts on each side
	trigger_if = {
		limit = {
			has_variable = qpq_ui_t_ekind
			has_variable = qpq_ui_i_ekind
		}
		custom_description = {
			text = QPQ_UI_FAIL_ROOM
			scope:actor = {
				OR = {
					NOT = { has_variable_list = qpq_pacts }
					variable_list_size = { name = qpq_pacts value < qpq_pact_cap_minus_one_value }
				}
			}
			scope:recipient = {
				OR = {
					NOT = { has_variable_list = qpq_pacts }
					variable_list_size = { name = qpq_pacts value < qpq_pact_cap_minus_one_value }
				}
			}
		}
	}
	trigger_else = {
		qpq_both_have_pact_capacity_trigger = { ACTOR = scope:actor RECIPIENT = scope:recipient }
	}
	custom_description = {
		text = QPQ_UI_FAIL_IMPRISONED
		OR = {
			var:qpq_ui_t_ekind ?= flag:clemency_bargain
			var:qpq_ui_i_ekind ?= flag:clemency_bargain
			AND = {
				is_imprisoned = no
				scope:recipient = { is_imprisoned = no }
			}
		}
	}
	custom_description = {
		text = QPQ_UI_FAIL_BUSY
		scope:recipient = {
			is_alive = yes
			NOT = { has_character_flag = qpq_considering_proposal }
		}
	}
	custom_description = {
		text = QPQ_UI_FAIL_REFUSE
		OR = {
			scope:recipient = { is_ai = no }
			AND = {
				exists = var:qpq_ui_acc_total
				var:qpq_ui_acc_total > 0
			}
		}
	}
}
