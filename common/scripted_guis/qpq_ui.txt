# Quid Pro Quo - Backroom Deal panel buttons (hand-written; the per-Ask buttons and pickers are in
# qpq_ui_kinds.txt). Root is always the player.

qpq_ui_close = {
	scope = character
	effect = {
		qpq_ui_close_effect = yes
	}
}

# Recompute the panel (tab changes): numbers go stale while the game runs
qpq_ui_refresh = {
	scope = character
	effect = {
		qpq_ui_refresh_effect = yes
	}
}

# Send the proposal
qpq_ui_propose = {
	scope = character
	saved_scopes = { actor recipient }

	is_valid = {
		qpq_ui_can_propose_trigger = yes
	}

	effect = {
		custom_tooltip = QPQ_UI_PROPOSE_TT
		hidden_effect = {
			qpq_ui_propose_effect = yes
		}
	}
}

# Command: your authority over a subordinate. Shown only for someone under you in your realm.
qpq_ui_command_shown = {
	scope = character
	saved_scopes = { actor recipient }
	is_shown = {
		qpq_command_possible_trigger = yes
	}
}

qpq_ui_command_0 = {
	scope = character
	effect = {
		set_variable = { name = qpq_ui_command value = 0 }
		qpq_ui_refresh_effect = yes
	}
}

qpq_ui_command_1 = {
	scope = character
	is_valid = {
		custom_description = {
			text = QPQ_UI_CMD_NEEDS_1
			exists = var:qpq_ui_cmd_margin

		}

		set_variable = { name = qpq_ui_command value = 1 }
