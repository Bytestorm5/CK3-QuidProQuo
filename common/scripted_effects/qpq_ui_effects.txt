# Quid Pro Quo - the Backroom Deal panel (gui/qpq_pact_window.gui)
#
# Root is always the player. The panel's state lives in variables on the player, so it is the same on
# every machine in multiplayer:
#   qpq_ui_open                     the window is shown
#   qpq_ui_recipient                who the deal is put to
#   qpq_ui_charter                  publish the deal as a Charter
# and, for each of the deal's two legs ("t": they do the deed and you give the goods; "i": you do the deed
# and they give the goods):
#   qpq_ui_<leg>_kind               the chosen deed (flag:none for goods only)
#   qpq_ui_<leg>_pick_char / _target_title / _target_artifact   the picked target
#   qpq_ui_<leg>_o_<good>, _timing, _qpq_x_<term>               the ticked boxes
# Everything else (tiers, prices, acceptance, candidate lists) is recomputed by qpq_ui_refresh_effect from
# those, by the generated helpers in qpq_ui_kind_effects.txt.
#
# scope:qpq_deal_me and scope:qpq_deal_them are the two characters. scope:actor and scope:recipient are
# whichever of them is the proposer of the Ask being worked on (see qpq_deal_orient_*_effect).

# The proposer is me
qpq_deal_orient_normal_effect = {
	scope:qpq_deal_me = { save_scope_as = actor }
	scope:qpq_deal_them = { save_scope_as = recipient }
}

# The proposer is them: they put the Ask to me
qpq_deal_orient_swapped_effect = {
	scope:qpq_deal_them = { save_scope_as = actor }
	scope:qpq_deal_me = { save_scope_as = recipient }
}

# Open the panel on RECIPIENT
qpq_ui_open_effect = {
	set_variable = { name = qpq_ui_recipient value = $RECIPIENT$ }
	qpq_ui_clear_choices_effect = yes
	qpq_ui_refresh_effect = yes
	set_variable = qpq_ui_open
}

qpq_ui_close_effect = {
	if = {
		limit = { has_variable = qpq_ui_open }
		remove_variable = qpq_ui_open
	}
	qpq_ui_clear_choices_effect = yes
	qpq_ui_clear_derived_effect = yes
}

# The two characters of the deal
qpq_ui_load_parties_effect = {
	save_scope_as = qpq_deal_me
	if = {
		limit = { exists = var:qpq_ui_recipient }
		var:qpq_ui_recipient = { save_scope_as = qpq_deal_them }
		qpq_deal_orient_normal_effect = yes
	}
}

# Recompute everything the panel shows
qpq_ui_refresh_effect = {
	qpq_ui_clear_derived_effect = yes
	qpq_ui_load_parties_effect = yes
	if = {
		limit = { exists = scope:qpq_deal_them }
		qpq_ui_leg_t_refresh_effect = yes
		qpq_ui_leg_i_refresh_effect = yes
		qpq_ui_deal_finish_effect = yes
	}
}

# Send the proposal. Everything is checked once more after a last refresh: if the world moved on
# while the panel was open, nothing is sent and the panel stays open showing what changed.
# AI recipients answer at once; a human recipient gets the proposal event.
qpq_ui_propose_effect = {
	qpq_ui_refresh_effect = yes
	if = {
		limit = { qpq_ui_can_propose_trigger = no }
		send_interface_toast = {
			type = event_toast_effect_neutral
			title = qpq_msg_proposal_changed_title
			left_icon = scope:qpq_deal_them
			custom_tooltip = qpq_msg_proposal_changed_tt
		}
	}
	else = {
		qpq_ui_save_deal_effect = yes
		if = {
			limit = { scope:qpq_deal_them = { is_ai = yes } }
			qpq_seal_deal_effect = yes
		}
		else = {
			scope:qpq_deal_them = {
				add_character_flag = { flag = qpq_considering_proposal days = 30 }
				trigger_event = qpq_deal.1
			}
			send_interface_toast = {
				type = event_toast_effect_neutral
				title = qpq_msg_proposal_sent_title
				left_icon = scope:qpq_deal_them
				custom_tooltip = qpq_msg_proposal_sent_tt
			}
		}
		qpq_ui_close_effect = yes
	}
}

# Seal the deal: each leg is a pact of the engine, and the two are tied to each other.
# Scopes: qpq_deal_me, qpq_deal_them, qpq_deal_<leg>_kind and the rest of what qpq_ui_save_deal_effect saved.
qpq_seal_deal_effect = {
	if = {
		limit = { exists = scope:qpq_deal_t_kind }
		qpq_deal_seal_leg_t_effect = yes
	}
	if = {
		limit = { exists = scope:qpq_deal_i_kind }
		qpq_deal_seal_leg_i_effect = yes
	}
	qpq_deal_link_effect = yes
	scope:qpq_deal_me = {
		if = {
			limit = { has_variable_list = qpq_deal_land_t }
			clear_variable_list = qpq_deal_land_t
		}
		if = {
			limit = { has_variable_list = qpq_deal_land_i }
			clear_variable_list = qpq_deal_land_i
		}
		if = {
			limit = { has_variable_list = qpq_deal_gift_artifact_t }
			clear_variable_list = qpq_deal_gift_artifact_t
		}
		if = {
			limit = { has_variable_list = qpq_deal_gift_artifact_i }
			clear_variable_list = qpq_deal_gift_artifact_i
		}
		if = {
			limit = { has_variable_list = qpq_deal_gift_secret_t }
			clear_variable_list = qpq_deal_gift_secret_t
		}
		if = {
			limit = { has_variable_list = qpq_deal_gift_secret_i }
			clear_variable_list = qpq_deal_gift_secret_i
		}
	}
}

# Tie the two pacts of a deal together: breaking one releases the other, and exposing one exposes both
qpq_deal_link_effect = {
	if = {
		limit = {
			exists = scope:qpq_deal_t_story
			exists = scope:qpq_deal_i_story
		}
		scope:qpq_deal_t_story = {
			set_variable = { name = qpq_mate value = scope:qpq_deal_i_story }
			qpq_deal_link_secrets_effect = { MATE = scope:qpq_deal_i_story }
		}
		scope:qpq_deal_i_story = {
			set_variable = { name = qpq_mate value = scope:qpq_deal_t_story }
			qpq_deal_link_secrets_effect = { MATE = scope:qpq_deal_t_story }
		}
	}
}

# Root: a pact story. Points both of its secrets at the first secret of the mate's pact.
qpq_deal_link_secrets_effect = {
	if = {
		limit = {
			exists = var:qpq_payer_secret
			exists = var:qpq_agent_secret
			exists = $MATE$.var:qpq_payer_secret
		}
		var:qpq_payer_secret = { set_variable = { name = qpq_mate_secret value = $MATE$.var:qpq_payer_secret } }
		var:qpq_agent_secret = { set_variable = { name = qpq_mate_secret value = $MATE$.var:qpq_payer_secret } }
	}
}
