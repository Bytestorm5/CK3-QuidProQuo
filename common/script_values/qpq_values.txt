# Quid Pro Quo - shared script values

# Pact cap: 2 + diplomacy / 5 active pacts per character, Charters included
qpq_pact_cap_value = {
	value = diplomacy
	divide = 5
	floor = yes
	add = 2
}

# Room for the two pacts of a deal that has a deed on each side: the cap, less one
qpq_pact_cap_minus_one_value = {
	value = qpq_pact_cap_value
	subtract = 1
}

# How the other character (scope:recipient) feels about the one proposing (scope:actor), in acceptance points
# Goodwill is worth only so much: past a point, being liked buys nothing more, and being hated costs only so much.
# It is never a refusal on its own, and it never pays for goods: see qpq_<kind>_goodwill_value.
qpq_opinion_points_value = {
	value = "scope:recipient.opinion(scope:actor)"
	multiply = 0.5
	min = -30
	max = 20
}

qpq_ui_deal_opinion_value = {
	value = qpq_opinion_points_value
}

############################################################
# Stake: what a deal is worth to the other side of the table. Scales with their rank and income, not
# the proposer's purse. Root: a character.
############################################################

qpq_stake_rank_value = {
	value = 0.5
	if = {
		limit = { is_landed = yes }
		value = 1
		if = {
			limit = { highest_held_title_tier >= tier_duchy }
			value = 2
		}
		if = {
			limit = { highest_held_title_tier >= tier_kingdom }
			value = 4
		}
		if = {
			limit = { highest_held_title_tier >= tier_empire }
			value = 8
		}
	}
	# A Head of Faith commands the Church's wealth: four times their rank, and never less than a lord's
	if = {
		limit = { qpq_is_head_of_faith_trigger = yes }
		multiply = 4
		min = 4
	}
}

qpq_stake_value = {
	value = monthly_character_income
	min = 8
	multiply = qpq_stake_rank_value
}

# A claim is worth the land it is on: the target's capital county (or the title named), by how developed it is, and
# nothing at all to someone who already has the claim. scope:actor gives it, scope:recipient gets it.
qpq_offer_claim_points_value = {
	value = 8
	if = {
		limit = { exists = scope:secondary_recipient }
		add = {
			value = 0
			if = {
				limit = { exists = scope:secondary_recipient.capital_county }
				scope:secondary_recipient.capital_county = {
					add = {
						value = development_level
						multiply = 0.8
						max = 14
					}
				}
			}
		}
	}
	if = {
		limit = { scope:recipient = { has_trait = ambitious } }
		add = 6
	}
	if = {
		limit = { scope:recipient = { has_trait = content } }
		add = -6
	}
	if = {
		limit = {
			exists = scope:secondary_recipient.capital_county
			scope:recipient = { has_claim_on = scope:secondary_recipient.capital_county }
		}
		multiply = 0
	}
	min = 0
}

# Effective tier of a half at the moment of exposure (root: the half)
qpq_effective_tier_value = {
	value = var:qpq_tier
	if = {
		limit = { has_variable = qpq_cover_up }
		add = 1
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = qpq_fresh }
				var:qpq_payer = { is_alive = no }
				var:qpq_agent = { is_alive = no }
			}
		}
		add = -1
	}
	if = {
		limit = { has_variable = qpq_victor }
		add = -1
		if = {
			limit = { var:qpq_owner = { is_independent_ruler = yes } }
			add = -1
		}
	}
	if = {
		limit = {
			exists = scope:qpq_exposer
			qpq_tolerates_intrigue_trigger = { CHAR = scope:qpq_exposer }
		}
		add = -1
	}
	min = 0
	max = 5
}

# Propose-a-Pact panel: order of the capped target lists. Rivals first, then people at either
# party's court or in their service, then everyone else; within each, the highest-ranked first.
# (scope:actor is the player, scope:recipient the other party)
qpq_ui_candidate_order_value = {
	value = highest_held_title_tier
	multiply = 10
	if = {
		limit = {
			OR = {
				has_relation_rival = scope:actor
				has_relation_rival = scope:recipient
			}
		}
		add = 1000
	}
	if = {
		limit = {
			OR = {
				is_courtier_of = scope:actor
				is_courtier_of = scope:recipient
				liege ?= scope:actor
				liege ?= scope:recipient

# What dirt on a person is worth to scope:recipient. Root: that person; scope:qpq_gift_payer knows the secrets.
# What it is worth depends on how bad the worst of it is for them, and on who they are to the one getting it.
qpq_gift_secret_points_value = {
	value = 8
	if = {
		limit = {
			any_secret = {
				is_known_by = scope:qpq_gift_payer
				NOT = { is_known_by = scope:recipient }
				is_criminal_for = secret_owner
			}
		}
		add = 20
	}
	else_if = {
		limit = {
			any_secret = {
				is_known_by = scope:qpq_gift_payer
				NOT = { is_known_by = scope:recipient }
				is_shunned_for = secret_owner
			}
		}
		add = 10
	}
	# How much their ruin would mean to the one who gets it
	if = {
		limit = {
			OR = {
				has_relation_rival = scope:recipient
				has_relation_nemesis = scope:recipient
			}
		}
		add = 14
	}
	if = {
		limit = { scope:recipient = { liege ?= prev } }
		add = 8
	}
	if = {
		limit = { liege ?= scope:recipient }
		add = 6
	}
	if = {
		limit = { is_close_family_of = scope:recipient }
		add = 4
	}
	# A secret on someone with nothing to lose is worth little
	if = {
		limit = { is_landed = no }
		multiply = 0.5
	}
	else_if = {
		limit = { highest_held_title_tier >= tier_kingdom }
		multiply = 1.4
	}
	round = yes
}

# The commander's standing: their legitimacy level, or their prestige level where they have no legitimacy
qpq_command_level_value = {
	value = 0
	scope:actor = {
		if = {
			limit = { has_legitimacy = yes }
			if = { limit = { legitimacy_level >= 1 } add = 1 }
			if = { limit = { legitimacy_level >= 2 } add = 1 }
			if = { limit = { legitimacy_level >= 3 } add = 1 }
			if = { limit = { legitimacy_level >= 4 } add = 1 }
			if = { limit = { legitimacy_level >= 5 } add = 1 }
		}
		else = {
			if = { limit = { prestige_level >= 1 } add = 1 }
			if = { limit = { prestige_level >= 2 } add = 1 }
			if = { limit = { prestige_level >= 3 } add = 1 }
			if = { limit = { prestige_level >= 4 } add = 1 }
			if = { limit = { prestige_level >= 5 } add = 1 }
		}
	}
}

# What the subordinate expects of the commander: vanilla's legitimacy expectation of a liege
# (default_ai_legitimacy_expectation, written here with explicit scopes), and one level more for each step
# further down the realm, since the writ of a distant lord carries less weight
qpq_command_expectation_value = {
	value = 1
	if = {
		limit = { scope:actor = { is_rightful_liege_of = scope:recipient } }
		subtract = 1
	}
	if = {
		limit = { scope:recipient = { is_powerful_vassal_of = scope:actor } }
		add = 1
	}
	if = {
		limit = { scope:recipient = { days_since_vassal_contract_liege_dynasty_reign_start >= 18250 } }
		subtract = 1
	}
	if = {
		limit = { scope:recipient = { days_since_vassal_contract_liege_dynasty_reign_start >= 36500 } }
		subtract = 1
	}
	if = {
		limit = { scope:recipient = { was_preferred_heir = scope:actor } }
		subtract = 1
	}
	if = {
		limit = {
			scope:actor = {
				OR = {
					is_spouse_of = scope:recipient
					is_close_family_of = scope:recipient
				}
			}
		}
		subtract = 1
	}
	if = {
		limit = { scope:actor.highest_held_title_tier >= tier_county }
		add = 1
	}
	if = {
		limit = { scope:actor.highest_held_title_tier >= tier_duchy }
		add = 1
	}
	if = {
		limit = { scope:actor.highest_held_title_tier >= tier_kingdom }
		add = 2
	}
	if = {
		limit = { scope:actor.highest_held_title_tier >= tier_empire }
		add = 2
	}
	if = {
		limit = { scope:actor.culture = { has_cultural_era_or_later = culture_era_tribal } }
		subtract = 1
	}
	else_if = {
		limit = { scope:actor.culture = { has_cultural_era_or_later = culture_era_high_medieval } }
		add = 2
	}
	min = 1
	max = 4
	# Deeper in the realm: a vassal's vassal expects more of you than of their own lord
	if = {
		limit = { NOT = { scope:recipient.liege ?= scope:actor } }
		add = 1
		if = {
			limit = { NOT = { scope:recipient.liege.liege ?= scope:actor } }
			add = 1
		}
	}
}

# How much root wants this character brought low: a rival, a claim on their land, root's opinion of them, faith
qpq_ai_enmity_value = {
	value = 0
	save_temporary_scope_as = qpq_ai_cand
	if = {
		limit = {
			OR = {
				has_relation_rival = root
				has_relation_nemesis = root
			}
		}
		add = 100
	}
	if = {
		limit = { root = { any_claim = { holder ?= scope:qpq_ai_cand } } }
		add = 40
	}
	add = {
		value = "root.opinion(scope:qpq_ai_cand)"
		multiply = -0.5
		min = -30
		max = 30
	}
	if = {
		limit = {
			root.faith = {
				faith_hostility_level = {
					target = scope:qpq_ai_cand.faith
					value >= faith_hostile_level
				}
			}
		}
		add = 15
	}
}

# How weak this character is next to the one who would fight them (scope:qpq_ai_fighter): the weaker the better,

		limit = { max_military_strength < { value = scope:qpq_ai_fighter.max_military_strength multiply = 0.5 } }

	}
	else_if = {
		limit = { max_military_strength < { value = scope:qpq_ai_fighter.max_military_strength multiply = 0.8 } }
		add = 25
	}
	else_if = {
		limit = { max_military_strength < scope:qpq_ai_fighter.max_military_strength }
		add = 10
	}
	else_if = {
		limit = { max_military_strength > { value = scope:qpq_ai_fighter.max_military_strength multiply = 2 } }
		add = -100
	}
	else_if = {
		limit = { max_military_strength > { value = scope:qpq_ai_fighter.max_military_strength multiply = 1.3 } }
		add = -40
	}
}
