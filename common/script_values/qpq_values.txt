# Quid Pro Quo - shared script values

# Pact cap: 2 + diplomacy / 5 active pacts per character, Charters included
qpq_pact_cap_value = {
	value = diplomacy
	divide = 5
	floor = yes
	add = 2
}

# How many pacts the character currently has running
qpq_active_pacts_value = {
	value = 0
	if = {
		limit = { has_variable_list = qpq_pacts }
		every_in_list = {
			variable = qpq_pacts
			add = 1
		}
	}
}

# Generic context shifts (see qpq_triggers.txt)
# Interaction scopes: scope:actor = payer, scope:recipient = agent,
# scope:secondary_recipient = target character, scope:target = target title

qpq_ctx_shift_payer_char_value = {
	value = 0
	if = {
		limit = { exists = scope:secondary_recipient }
		if = {
			limit = { qpq_ctx_hostile_foreigner_trigger = { PARTY = scope:actor TARGET = scope:secondary_recipient } }
			add = -1
		}
		if = {
			limit = { qpq_ctx_coreligionist_ally_trigger = { PARTY = scope:actor TARGET = scope:secondary_recipient } }
			add = 1
		}
		if = {
			limit = { qpq_ctx_same_realm_trigger = { PARTY = scope:actor TARGET = scope:secondary_recipient } }
			add = 1
		}
		if = {
			limit = { qpq_ctx_close_kin_trigger = { PARTY = scope:actor TARGET = scope:secondary_recipient } }
			add = 1
		}
		if = {
			limit = { qpq_ctx_own_liege_trigger = { PARTY = scope:actor TARGET = scope:secondary_recipient } }
			add = 2
		}
		if = {
			limit = { qpq_ctx_clergy_trigger = { TARGET = scope:secondary_recipient } }
			add = 1
		}
	}
}

	value = scope:qpq_f_mult
	divide = 100

	else_if = { limit = { scope:qpq_f_tier = 4 } add = 500 }
	else_if = { limit = { scope:qpq_f_tier >= 5 } add = 800 }
	multiply = qpq_f_scale_value
	multiply = -1
	round = yes

# AI-initiated pacts: yearly chance (percent) for a character to propose one.
# "Slightly rarer than uncommon", higher for intrigue-heavy characters.
qpq_ai_proposal_chance_value = {
	value = 3
	if = {
		limit = { has_game_rule = qpq_ai_pacts_frequent }
		multiply = 2
	}
	else_if = {
		limit = { has_game_rule = qpq_ai_pacts_rare }
		multiply = 0.4
	}
	if = {
		limit = { intrigue >= 14 }
		multiply = 1.75
	}
	if = {
		limit = { has_trait = schemer }
		multiply = 1.5
	}
	if = {
		limit = { has_trait = deceitful }
		multiply = 1.3
	}
	if = {
		limit = { has_lifestyle = intrigue_lifestyle }
		multiply = 1.3
	}
	if = {
		limit = {
			OR = {
				has_trait = honest
				has_trait = just
			}
		}
		multiply = 0.5
	}
}

# Exposure pricing (root: a pact half)
############################################################

# Effective tier of a half at the moment of exposure (root: the half)
qpq_effective_tier_value = {
	value = var:qpq_tier
	if = {
		limit = { has_variable = qpq_cover_up }
		add = 1
	}
	if = {
		limit = {
			OR = {
				NOT = { has_variable = qpq_fresh }
				var:qpq_payer = { is_alive = no }
				var:qpq_agent = { is_alive = no }
			}
		}
		add = -1
	}
	if = {
		limit = { has_variable = qpq_victor }
		add = -1
		if = {
			limit = { var:qpq_owner = { is_independent_ruler = yes } }
			add = -1
		}
	}
	if = {
		limit = {
			exists = scope:qpq_exposer
			qpq_tolerates_intrigue_trigger = { CHAR = scope:qpq_exposer }
		}
		add = -1
	}
	min = 0
	max = 5
}

# Share of the baseline cost this half's owner pays (0-100)
qpq_payer_share_value = {
	value = 100
	if = {
		limit = {
			exists = var:qpq_confessed_by
			var:qpq_confessed_by = var:qpq_owner
		}
		value = 45
		if = {
			limit = { has_variable = qpq_deathbed }
			value = 15
		}
	}
	else_if = {
		limit = { scope:qpq_exposer = var:qpq_owner }
		value = 60
		if = {
			limit = { var:qpq_wronged ?= var:qpq_owner }
			multiply = 0.25
		}
	}
}
